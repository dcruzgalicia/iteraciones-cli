\renewcommand*{\dictumrule}{}
\renewcommand*{\dictumauthorformat}[1]{#1}
\newlength{\alturaA}
\settoheight{\alturaA}{\hbox{Á}}
\makeatletter
\renewcommand{\dictum}[2][]{\par%
  \begingroup
    \raggeddictum\parbox[t]{\dictumwidth}{%
      {\@dictumfont{\raggeddictumtext #2\strut\par}%
        \def\@tempa{#1}\ifx\@tempa\@empty\else%
          {\raggeddictumtext\dictumrule}%
          \raggeddictumauthor\@dictumauthorfont\dictumauthorformat{#1}%
          \strut\par%
        \fi%
      }%
    }\par%
  \endgroup
}
\makeatother
