% Endpapers (guardas): imagen de fondo SOLO en la página 1 (la hoja de guarda
% en blanco que se inserta antes de la página de extratitle, que está en la
% 3): en ninguna otra página. La imagen se mide una sola vez en \setendpapers
% con un \sbox (a tamaño natural: \wd = ancho, \ht = alto).
%
% Modo COVER centrado: la imagen cubre la hoja completa sin espacios en
% blanco ni deformación. El LADO MÁS CORTO de la imagen define la dimensión
% fija = papel + 6mm (el ancho si la imagen es más alta, el alto si es más
% ancha); la otra dimensión crece según la proporción y sobra por ambos
% lados. El centro de la imagen se alinea con el centro de la hoja: el \put
% en (.5\paperwidth, .5\paperheight) cae en el centro con y sin el grid de
% 24-eso-pic (el origen del \put se invierte entre ambos, pero la mitad del
% papel desde cualquier borde es el centro). La caja 0×0 con \hss/\vss centra
% la imagen sobre ese punto con el sobrante simétrico.
% El tamaño de la hoja se toma de \paperwidth/\paperheight (documentclass o
% geometry, incluido el override del proyecto): no se hardcodea.
% Los comandos no pueden llamarse \endpapers: LaTeX prohíbe nombres que
% empiecen con \end (colisión con el primitivo \end).
\usepackage{eso-pic}
\makeatletter
\newsavebox{\papersbox}
\newcommand*{\@endpapers}{}%
\newcommand{\setendpapers}[1]{%
  \gdef\@endpapers{#1}%
  \sbox{\papersbox}{\includegraphics{#1}}%
}%
\newcommand{\drawendpapers}{%
  % Solo la página 1 Y solo si existen las hojas de guarda (\titlepageguards,
  % activada por \titlepageblanks en 19-maketitle.tex): el endpaper es el
  % fondo de la hoja en blanco; sin guardas, no se agrega en ningún caso.
  \ifnum\value{page}=1
    \iftitlepageguards
      \ifx\@endpapers\@empty
      \else
        % Modo COVER centrado: el centro de la imagen se alinea con el centro
        % de la hoja. El \put x = .5\paperwidth vale en ambos casos; el y
        % depende del grid de 24-eso-pic (desplaza el origen del \put al
        % borde superior): sin grid +.5\paperheight, con grid -.5\paperheight.
        % La caja 0×0 con \hss/\vss centra la imagen sobre ese punto con el
        % sobrante simétrico (cubre la hoja sin espacios en blanco).
        \put(.5\paperwidth,\ifx\ESO@HookIIIBG\@empty .5\paperheight\else -.5\paperheight\fi){%
          \vbox to 0pt{%
            \vss
            \hbox to 0pt{%
              \hss
              \ifdim\wd\papersbox<\ht\papersbox
                \includegraphics[width=\dimexpr\paperwidth+6mm\relax,keepaspectratio]{\@endpapers}%
              \else
                \includegraphics[height=\dimexpr\paperheight+6mm\relax,keepaspectratio]{\@endpapers}%
              \fi
              \hss
            }%
            \vss
          }%
        }%
      \fi
    \fi
  \fi
}
\AddToShipoutPictureBG*{\drawendpapers}
\makeatother
