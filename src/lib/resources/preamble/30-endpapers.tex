% Endpapers (guardas): imagen de fondo que cubre la hoja completa SOLO en la
% página 1 (la hoja de guarda en blanco que se inserta antes de la página de
% extratitle, que está en la 3): en ninguna otra página. La imagen se mide una
% sola vez en \setendpapers con un \sbox (a tamaño natural: \wd = ancho,
% \ht = alto) y el LADO MÁS CORTO decide la única dimensión fija: papel +
% 6mm. La otra dimensión sigue la proporción (keepaspectratio): la imagen
% cubre la hoja sin deformarse. El tamaño de la hoja se toma de
% \paperwidth/\paperheight (documentclass o geometry, incluido el override
% del proyecto): no se hardcodea. \put(-4mm,-3mm) deja un sangrado de ~2-3mm
% en los bordes derecho e inferior (el +6mm compensa el offset).
%
% Compatibilidad con 24-eso-pic: el grid extiende el picture del shipout
% hasta -\paperheight y desplaza el origen del \put (el endpaper quedaba fuera
% de la hoja cuando el grid estaba activo). \ESO@HookIIIBG no vacío indica que
% el grid está presente: en ese caso se ancla al fondo con \AtPageLowerLeft;
% sin grid, el \put se mide desde el fondo de la misma forma.
% Los comandos no pueden llamarse \endpapers: LaTeX prohíbe nombres que
% empiecen con \end (colisión con el primitivo \end).
\usepackage{eso-pic}
\makeatletter
\newsavebox{\papersbox}
\newcommand*{\@endpapers}{}%
\newcommand{\setendpapers}[1]{%
  \gdef\@endpapers{#1}%
  \sbox{\papersbox}{\includegraphics{#1}}%
}%
\newcommand{\drawendpapers}{%
  % Solo la página 1 Y solo si existen las hojas de guarda (\titlepageguards,
  % activada por \titlepageblanks en 19-maketitle.tex): el endpaper es el
  % fondo de la hoja en blanco; sin guardas, no se agrega en ningún caso.
  \ifnum\value{page}=1
    \iftitlepageguards
      \ifx\@endpapers\@empty
      \else
        \ifx\ESO@HookIIIBG\@empty
          \put(-4mm,-3mm){%
            \ifdim\wd\papersbox<\ht\papersbox
              \includegraphics[width=\dimexpr\paperwidth+6mm\relax,keepaspectratio]{\@endpapers}%
            \else
              \includegraphics[height=\dimexpr\paperheight+6mm\relax,keepaspectratio]{\@endpapers}%
            \fi
          }%
        \else
          \AtPageLowerLeft{%
            \put(-4mm,-3mm){%
              \ifdim\wd\papersbox<\ht\papersbox
                \includegraphics[width=\dimexpr\paperwidth+6mm\relax,keepaspectratio]{\@endpapers}%
              \else
                \includegraphics[height=\dimexpr\paperheight+6mm\relax,keepaspectratio]{\@endpapers}%
              \fi
            }%
          }%
        \fi
      \fi
    \fi
  \fi
}
\AddToShipoutPictureBG*{\drawendpapers}
\makeatother
