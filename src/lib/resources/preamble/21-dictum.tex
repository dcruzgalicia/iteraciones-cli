\setkomafont{dictum}{\normalsize\normalfont\itshape}
\setkomafont{dictumauthor}{\normalsize\normalfont}
\renewcommand*{\dictumwidth}{0.5\textwidth}
\renewcommand*{\dictumrule}{}
\renewcommand*{\dictumauthorformat}[1]{#1}
\renewcommand*{\raggeddictumtext}{}
\makeatletter
\renewcommand*{\raggeddictum}{%
  \leftmargin=\dimexpr\linewidth-\dictumwidth\relax
  \rightmargin=\z@
}
\renewcommand{\dictum}[3][]{%
  \par
  \@nobreakfalse
  \begingroup
    \ifx\relax#3\relax\else\renewcommand{\dictumwidth}{#3\textwidth}\fi
    \list{}{%
      \itemsep=\z@
      \parsep=\z@
      \partopsep=\z@
      \topsep=2\baselineskip
      \raggeddictum
      \listparindent=\z@
      \itemindent=\z@
    }%
    \item\relax
    \@dictumfont
    {%
      \raggeddictumtext
      #2\strut\par
    }%
    \def\@tempa{#1}%
    \ifx\@tempa\@empty
    \else
      {%
        \raggeddictumtext
        \dictumrule
      }%
      \raggeddictumauthor
      \@dictumauthorfont
      \dictumauthorformat{#1}%
      \strut\par
    \fi
    \endlist
  \endgroup
}
% La misma cita, pero para contenido con bloques dentro (un shrink, por
% ejemplo): el macro pide el texto como argumento y un bloque con párrafos no
% cabe ahí.
\newenvironment{dictumbloque}[2][]{%
  \par
  \@nobreakfalse
  \begingroup
    \def\dictumautor{#1}%
    \ifx\relax#2\relax\else\renewcommand{\dictumwidth}{#2\textwidth}\fi
    \list{}{%
      \itemsep=\z@
      \parsep=\z@
      \partopsep=\z@
      \topsep=2\baselineskip
      \raggeddictum
      \listparindent=\z@
      \itemindent=\z@
    }%
    \item\relax
    \@dictumfont
    \raggeddictumtext
}{%
    \edef\@tempa{\dictumautor}%
    \ifx\@tempa\@empty
    \else
      {%
        \raggeddictumtext
        \dictumrule
      }%
      \raggeddictumauthor
      \@dictumauthorfont
      \dictumauthorformat{\dictumautor}%
      \strut\par
    \fi
    \endlist
  \endgroup
}
\makeatother
