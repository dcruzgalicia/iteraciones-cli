% pandoc 3.2.1+ envuelve \includegraphics en \pandocbounded para imágenes
% sin width/height explícito. Muestra la imagen a tamaño natural pero la
% escala si excede el ancho (\linewidth) o alto (\textheight) del texto.
% Dependencia: graphicx (cargado en 19-maketitle.tex).
\makeatletter
\newsavebox\pandoc@box
\newcommand*\pandocbounded[1]{%
  \sbox\pandoc@box{#1}%
  \Gscale@div\@tempa{\textheight}{\dimexpr\ht\pandoc@box+\dp\pandoc@box\relax}%
  \Gscale@div\@tempb{\linewidth}{\wd\pandoc@box}%
  \ifdim\@tempb\p@<\@tempa\p@\let\@tempa\@tempb\fi
  \ifdim\@tempa\p@<\p@\scalebox{\@tempa}{\usebox\pandoc@box}%
  \else\usebox{\pandoc@box}%
  \fi
}
\makeatother
